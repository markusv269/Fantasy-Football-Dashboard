name: Weekly League Avatar Sync

# League avatars rarely change, so a weekly pass is enough. This workflow
# updates only the `avatar` column of existing `leagues` rows using the
# existing `app/scripts/sync_league_avatars.py` script — it never inserts new
# rows and never overwrites a populated avatar with an empty Sleeper value
# unless `--clear-empty` is explicitly requested via manual dispatch.

on:
  schedule:
    - cron: "0 4 * * 1"  # Mondays at 04:00 UTC
  workflow_dispatch:
    inputs:
      clear_empty:
        description: "If Sleeper returns an empty avatar, clear the DB value (default: keep existing)"
        required: false
        default: "false"
        type: choice
        options:
          - "true"
          - "false"
      dry_run:
        description: "Log intended updates without writing to Supabase"
        required: false
        default: "false"
        type: choice
        options:
          - "true"
          - "false"

concurrency:
  group: sync-avatars-weekly
  cancel-in-progress: false

jobs:
  sync:
    name: Refresh league avatars
    runs-on: ubuntu-latest
    timeout-minutes: 60
    env:
      SUPABASE_URL: ${{ secrets.SUPABASE_URL }}
      SUPABASE_KEY: ${{ secrets.SUPABASE_KEY }}
      PYTHONUNBUFFERED: "1"
    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: "3.13"
          cache: "pip"

      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          pip install -r requirements.txt

      - name: Verify Supabase credentials are present
        run: |
          if [ -z "${SUPABASE_URL}" ] || [ -z "${SUPABASE_KEY}" ]; then
            echo "::error::SUPABASE_URL and SUPABASE_KEY secrets must be configured on this repository."
            exit 2
          fi
          echo "Supabase credentials are set."

      - name: Sync league avatars
        run: |
          FLAGS=""
          if [ "${{ github.event.inputs.clear_empty }}" = "true" ]; then
            FLAGS="${FLAGS} --clear-empty"
          fi
          if [ "${{ github.event.inputs.dry_run }}" = "true" ]; then
            FLAGS="${FLAGS} --dry-run"
          fi
          python -m app.scripts.sync_league_avatars ${FLAGS}
