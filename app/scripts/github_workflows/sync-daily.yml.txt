name: Daily Full Sync (metadata, managers, rosters, drafts)

# Once-a-day full pass across every league in Supabase. Includes:
#   - league metadata (preserving league_type)
#   - manager rows (users + roster owners)
#   - roster rows for the current NFL week (with players/starters/reserve/taxi)
#   - matchup_week_stats for weeks 1..current
#   - drafts (metadata) for each league
#
# Runs unconditionally every day so offseason schedule changes, new drafts and
# manager updates land within 24 hours.

on:
  schedule:
    - cron: "0 6 * * *"  # 06:00 UTC daily
  workflow_dispatch:
    inputs:
      skip_matchups:
        description: "Skip matchup week upserts"
        required: false
        default: "false"
        type: choice
        options:
          - "true"
          - "false"
      skip_drafts:
        description: "Skip draft metadata upserts"
        required: false
        default: "false"
        type: choice
        options:
          - "true"
          - "false"

concurrency:
  group: sync-daily
  cancel-in-progress: false

jobs:
  sync:
    name: Daily full league sync
    runs-on: ubuntu-latest
    timeout-minutes: 340
    env:
      SUPABASE_URL: ${{ secrets.SUPABASE_URL }}
      SUPABASE_KEY: ${{ secrets.SUPABASE_KEY }}
      PYTHONUNBUFFERED: "1"
    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: "3.13"
          cache: "pip"

      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          pip install -r requirements.txt

      - name: Verify Supabase credentials are present
        run: |
          if [ -z "${SUPABASE_URL}" ] || [ -z "${SUPABASE_KEY}" ]; then
            echo "::error::SUPABASE_URL and SUPABASE_KEY secrets must be configured on this repository."
            exit 2
          fi
          echo "Supabase credentials are set."

      - name: Run daily full sync
        run: |
          FLAGS=""
          if [ "${{ github.event.inputs.skip_matchups }}" = "true" ]; then
            FLAGS="${FLAGS} --skip-matchups"
          fi
          if [ "${{ github.event.inputs.skip_drafts }}" = "true" ]; then
            FLAGS="${FLAGS} --skip-drafts"
          fi
          python -m app.scripts.weekly_sync ${FLAGS}
