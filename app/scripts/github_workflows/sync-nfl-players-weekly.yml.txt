name: Weekly NFL Player Catalog Sync

# Refreshes the `nfl_players` table with the full Sleeper NFL catalog
# (~11k rows). The catalog changes slowly — new rookies, trades, waivers —
# so weekly is enough. Runs on Tuesdays at 03:00 UTC to avoid clashing
# with the daily full workflow.

on:
  schedule:
    - cron: "0 3 * * 2"  # Tuesdays at 03:00 UTC
  workflow_dispatch: {}

concurrency:
  group: sync-nfl-players-weekly
  cancel-in-progress: false

jobs:
  sync:
    name: Refresh NFL player catalog
    runs-on: ubuntu-latest
    timeout-minutes: 60
    env:
      SUPABASE_URL: ${{ secrets.SUPABASE_URL }}
      SUPABASE_KEY: ${{ secrets.SUPABASE_KEY }}
      PYTHONUNBUFFERED: "1"
    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: "3.13"
          cache: "pip"

      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          pip install -r requirements.txt

      - name: Verify Supabase credentials are present
        run: |
          if [ -z "${SUPABASE_URL}" ] || [ -z "${SUPABASE_KEY}" ]; then
            echo "::error::SUPABASE_URL and SUPABASE_KEY secrets must be configured on this repository."
            exit 2
          fi
          echo "Supabase credentials are set."

      - name: Sync full NFL player catalog
        run: python -m app.scripts.targeted_sync --modes nfl_players
